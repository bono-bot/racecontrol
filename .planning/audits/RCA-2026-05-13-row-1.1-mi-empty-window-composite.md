```rust
// crates/racecontrol/src/mi/empty_window_detector.rs (NEW)
pub fn spawn_empty_window_detector(state: Arc<AppState>) {
    tokio::spawn(async move {
        let mut interval = tokio::time::interval(Duration::from_secs(60));
        loop {
            interval.tick().await;
            if let Err(e) = detect_and_emit(&state).await {
                tracing::error!("[mi] empty_window detect: {}", e);
            }
        }
    });
}
async fn detect_and_emit(state: &Arc<AppState>) -> Result<()> {
    let now = Utc::now();
    let window_start = now - Duration::minutes(5);
    let snapshot = compute_occupancy_window(state, window_start, now).await?;
    if snapshot.occupancy_pct < 0.25 {  // empirically-tuned threshold (Wave 4 calibration)
        let event = MiEmptyWindowEvent::from_snapshot(snapshot);
        emit_mi_event(state, event).await?;
    }
    Ok(())
}
```
