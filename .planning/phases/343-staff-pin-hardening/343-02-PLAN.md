Purpose: The Vishal incident showed HTTP 200 is not evidence a PIN change worked. Two-layer verification (immediate + delayed) catches both direct-write bugs and sync-overwrite bugs.

**post_write_verify helper signature:**
```rust
async fn post_write_verify_staff_pin(
    state: &Arc<AppState>,
    staff_id: &str,
    expected_pin: &str,
) -> Result<(), String> {
    let row = sqlx::query_scalar::<_, String>(
        "SELECT pin FROM staff_members WHERE id = ?"
    ).bind(staff_id).fetch_optional(&state.db).await
        .map_err(|e| format!("verify query failed: {}", e))?;
    match row {
        Some(actual) if actual == expected_pin => Ok(()),
        Some(actual) => Err(format!("pin mismatch: expected={} actual={}", expected_pin, actual)),
        None => Err("staff row not found after update".to_string()),
    }
}
```

**Delayed sync verify (spawned task):**
```rust
fn spawn_delayed_sync_verify(
    state: Arc<AppState>,
    staff_id: String,
    expected_pin: String,
    correlation_id: String,
) {
    let sync_interval = state.config.cloud.sync_interval_secs.unwrap_or(30);
    let delay = Duration::from_secs(sync_interval + 5);
    tokio::spawn(async move {
        tokio::time::sleep(delay).await;
        match post_write_verify_staff_pin(&state, &staff_id, &expected_pin).await {
            Ok(()) => {
                tracing::info!(
                    target: "auth",
                    correlation_id = %correlation_id,
                    staff_id = %staff_id,
                    "Staff PIN delayed verify passed (survived one sync cycle)"
                );
            }
            Err(e) => {
                tracing::error!(
                    target: "auth",
                    correlation_id = %correlation_id,
                    staff_id = %staff_id,
                    error = %e,
                    "Staff PIN delayed verify FAILED — sync may have reverted the change"
                );
                let _ = sqlx::query(
                    "INSERT INTO alert_incidents (id, severity, source, message, metadata, created_at)
                     VALUES (?, 'P1', 'staff_pin_verify', ?, ?, datetime('now'))"
                )
                .bind(format!("pin_revert_{}", correlation_id))
                .bind(format!("Staff PIN change reverted for {}: {}", staff_id, e))
                .bind(format!(r#"{{"staff_id":"{}","correlation_id":"{}"}}"#, staff_id, correlation_id))
                .execute(&state.db)
                .await;
                // TODO: WhatsApp alert (deferred — depends on whatsapp_alerter refactor)
            }
        }
    });
}
```

**Integration into update_staff (after successful UPDATE):**
```rust
if let Some(ref new_pin) = req.pin {
    let correlation_id = uuid::Uuid::new_v4().to_string();
    // Immediate verify
    if let Err(e) = post_write_verify_staff_pin(&state, &id, new_pin).await {
        tracing::error!(target: "auth", correlation_id = %correlation_id, error = %e, "post_write_verify failed");
        return (StatusCode::INTERNAL_SERVER_ERROR, Json(json!({
            "error": format!("PIN write verification failed: {}", e),
            "correlation_id": correlation_id,
        })));
    }
    // Delayed verify
    spawn_delayed_sync_verify(state.clone(), id.clone(), new_pin.clone(), correlation_id.clone());
    return (StatusCode::OK, Json(json!({
        "id": id,
        "status": "ok",
        "verified": true,
        "correlation_id": correlation_id,
    })));
}
```
