The test: creates a throwaway staff record, changes the PIN via the new admin endpoint, waits more than one sync interval, and re-validates — catching any future bug where cloud sync reverts a PIN change.

**Test flow (Playwright style):**
```ts
import { test, expect } from '@playwright/test';
import { ApiClient } from '../../fixtures/api-client';
import { getAdminToken } from '../../fixtures/auth';

const SYNC_WAIT_MS = 70 * 1000; // sync_interval_secs(30) * 2 + 10s safety

test.describe('Staff PIN lifecycle (silent-revert regression)', () => {
  let adminToken: string;
  let testStaffId: string;
  const TEST_PIN_INITIAL = '9999';
  const TEST_PIN_CHANGED = '8888';
  const TEST_NAME = `TEST_PIN_LIFECYCLE_${Date.now()}`;

  test.beforeAll(async () => {
    adminToken = await getAdminToken();
  });

  test('1. Create test staff with PIN 9999', async () => {
    const api = new ApiClient(process.env.CLOUD_API_URL!, adminToken);
    const r = await api.post('/staff', { name: TEST_NAME, pin: TEST_PIN_INITIAL, phone: '' });
    expect(r.status).toBe(200);
    expect(r.body.id).toMatch(/^staff_/);
    testStaffId = r.body.id;
  });

  test('2. Validate PIN 9999 on cloud', async () => {
    const api = new ApiClient(process.env.CLOUD_API_URL!);
    const r = await api.post('/staff/validate-pin', { pin: TEST_PIN_INITIAL });
    expect(r.status).toBe(200);
    expect(r.body.staff_id).toBe(testStaffId);
  });

  test('3. Wait for cloud→venue sync, then validate PIN 9999 on venue', async () => {
    await new Promise(r => setTimeout(r, 35_000));
    const api = new ApiClient(process.env.VENUE_API_URL!);
    const r = await api.post('/staff/validate-pin', { pin: TEST_PIN_INITIAL });
    expect(r.status).toBe(200);
    expect(r.body.staff_id).toBe(testStaffId);
  });

  test('4. Change PIN 9999 → 8888 via admin endpoint', async () => {
    const api = new ApiClient(process.env.CLOUD_API_URL!, adminToken);
    const r = await api.post(`/admin/staff/${testStaffId}/change-pin`, { new_pin: TEST_PIN_CHANGED });
    expect(r.status).toBe(200);
    expect(r.body.cloud_verified).toBe(true);
    expect(r.body.venue_verified).toBe(true);
    expect(r.body.correlation_id).toBeTruthy();
  });

  test('5. Immediately (t+2s) validate PIN 8888 on cloud + venue', async () => {
    await new Promise(r => setTimeout(r, 2000));
    const cloud = new ApiClient(process.env.CLOUD_API_URL!);
    const venue = new ApiClient(process.env.VENUE_API_URL!);
    expect((await cloud.post('/staff/validate-pin', { pin: TEST_PIN_CHANGED })).status).toBe(200);
    expect((await venue.post('/staff/validate-pin', { pin: TEST_PIN_CHANGED })).status).toBe(200);
  });

  test('6. Wait SYNC_WAIT_MS, re-validate PIN 8888 — catches silent revert', async () => {
    await new Promise(r => setTimeout(r, SYNC_WAIT_MS));
    const cloud = new ApiClient(process.env.CLOUD_API_URL!);
    const venue = new ApiClient(process.env.VENUE_API_URL!);
    const cr = await cloud.post('/staff/validate-pin', { pin: TEST_PIN_CHANGED });
    const vr = await venue.post('/staff/validate-pin', { pin: TEST_PIN_CHANGED });
    expect(cr.status).toBe(200);
    expect(vr.status).toBe(200);
    // This is the assertion that would have failed on 2026-04-09
  });

  test('7. Old PIN 9999 no longer works', async () => {
    const cloud = new ApiClient(process.env.CLOUD_API_URL!);
    const venue = new ApiClient(process.env.VENUE_API_URL!);
    expect((await cloud.post('/staff/validate-pin', { pin: TEST_PIN_INITIAL })).status).toBe(401);
    expect((await venue.post('/staff/validate-pin', { pin: TEST_PIN_INITIAL })).status).toBe(401);
  });

  test.afterAll(async () => {
    if (!testStaffId) return;
    const api = new ApiClient(process.env.CLOUD_API_URL!, adminToken);
    await api.delete(`/staff/${testStaffId}`);
    // Verify delete propagated
    await new Promise(r => setTimeout(r, 35_000));
    const venue = new ApiClient(process.env.VENUE_API_URL!);
    const r = await venue.post('/staff/validate-pin', { pin: TEST_PIN_CHANGED });
    expect(r.status).toBe(401); // deleted staff cannot login
  });
});
```
