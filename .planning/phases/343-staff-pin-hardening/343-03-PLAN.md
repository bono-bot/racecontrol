**Orchestration pseudocode:**
```rust
async fn change_staff_pin_safe(id, new_pin, admin_jwt) -> Response {
    let correlation_id = Uuid::new_v4();
    let started = Instant::now();

    // Step 1: Write to cloud (if venue, forward; if cloud, write direct)
    let write_result = if this_instance_is_cloud() {
        direct_update_staff(id, new_pin).await
    } else {
        forward_to_cloud(id, new_pin, admin_jwt).await
    };
    // ... handle errors ...

    // Step 2: Force sync on venue if we wrote to cloud
    if !this_instance_is_cloud() {
        return_err; // we're not cloud, user should hit cloud admin page
    }
    // We ARE cloud here, so venue needs a pull
    trigger_venue_sync_via_cloud_relay(vec!["staff_members"]).await;

    // Step 3: Verify on cloud (direct DB read)
    let cloud_verified = post_write_verify_staff_pin(&state, id, new_pin).await.is_ok();

    // Step 4: Verify on venue (HTTP validate-pin round-trip)
    let venue_verified = venue_validate_pin_http(id, new_pin).await.is_ok();

    Json(json!({
        "status": if cloud_verified && venue_verified { "ok" } else { "partial" },
        "staff_id": id,
        "cloud_verified": cloud_verified,
        "venue_verified": venue_verified,
        "latency_ms": started.elapsed().as_millis(),
        "correlation_id": correlation_id.to_string(),
    }))
}
```

**Admin Next.js route: racingpoint-admin/src/app/api/admin/staff/[id]/change-pin/route.ts**
```ts
export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const { new_pin } = await req.json();
  const adminJwt = req.cookies.get('admin_jwt')?.value;
  if (!adminJwt) return NextResponse.json({error:"unauth"}, {status:401});

  const cloudUrl = process.env.NEXT_PUBLIC_CLOUD_API_URL || "https://api.racingpoint.cloud/api/v1";
  const r = await fetch(`${cloudUrl}/admin/staff/${params.id}/change-pin`, {
    method: "POST",
    headers: { "Content-Type": "application/json", "Authorization": `Bearer ${adminJwt}` },
    body: JSON.stringify({ new_pin }),
  });
  const body = await r.json();
  return NextResponse.json(body, { status: r.status });
}
```

    ```rust
    #[derive(Deserialize)]
    struct SyncPullNowRequest { tables: Vec<String> }

    async fn sync_pull_now(
        State(state): State<Arc<AppState>>,
        Json(req): Json<SyncPullNowRequest>,
    ) -> (StatusCode, Json<Value>) {
        let started = Instant::now();
        let mut results = Vec::new();
        for table in &req.tables {
            match cloud_sync::pull_single_table(&state, table).await {
                Ok(n) => results.push(json!({"table": table, "rows": n, "ok": true})),
                Err(e) => results.push(json!({"table": table, "ok": false, "error": e.to_string()})),
            }
        }
        (StatusCode::OK, Json(json!({
            "ok": true,
            "tables": results,
            "duration_ms": started.elapsed().as_millis(),
        })))
    }
    ```
