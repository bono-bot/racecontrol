    ```rust
    use std::collections::HashMap;
    use std::sync::Arc;
    use tokio::sync::RwLock;
    use chrono::Utc;
    use rc_common::protocol::{LaunchStatusCard, LaunchState, LaunchOrigin};

    const MAX_ACTIVE_LAUNCHES: usize = 100;
    const STALE_AFTER_SECS: i64 = 600; // 10 min

    pub struct LaunchStateMachine {
        states: RwLock<HashMap<String, LaunchStatusCard>>,
    }

    impl LaunchStateMachine {
        pub fn new() -> Self {
            Self { states: RwLock::new(HashMap::new()) }
        }

        pub async fn start_launch(
            &self,
            launch_id: String,
            pod_id: String,
            sim_type: String,
            origin: LaunchOrigin,
        ) -> LaunchStatusCard {
            let card = LaunchStatusCard {
                launch_id: launch_id.clone(),
                pod_id,
                sim_type,
                state: LaunchState::LaunchStarted,
                detail: None,
                timestamp: Utc::now(),
                origin,
                ai_tier: None,
                fix_action: None,
            };
            // CLAUDE.md rule: lock-and-drop BEFORE any await
            {
                let mut guard = self.states.write().await;
                guard.insert(launch_id, card.clone());
            }
            card
        }

        pub async fn transition(
            &self,
            launch_id: &str,
            new_state: LaunchState,
            detail: Option<String>,
            ai_tier: Option<u8>,
            fix_action: Option<String>,
        ) -> Option<LaunchStatusCard> {
            let updated = {
                let mut guard = self.states.write().await;
                let existing = guard.get_mut(launch_id)?;
                if !Self::is_valid_transition(existing.state, new_state) {
                    return None;
                }
                existing.state = new_state;
                if detail.is_some() { existing.detail = detail; }
                if ai_tier.is_some() { existing.ai_tier = ai_tier; }
                if fix_action.is_some() { existing.fix_action = fix_action; }
                existing.timestamp = Utc::now();
                existing.clone()
            }; // guard dropped here — safe to await below
            Some(updated)
        }

        fn is_valid_transition(from: LaunchState, to: LaunchState) -> bool {
            use LaunchState::*;
            // Terminal states: no further transitions
            if matches!(from, IssueFixed | NeedsManualIntervention) {
                return false;
            }
            // Monotonic forward order
            match (from, to) {
                (LaunchStarted, AiAnalysisRequested) => true,
                (LaunchStarted, IssueFixed) => true,  // happy path skip
                (LaunchStarted, NeedsManualIntervention) => true, // billing-reject path — see Plan 01 Task 3 Step C
                (AiAnalysisRequested, IssueBeingFixed) => true,
                (AiAnalysisRequested, NeedsManualIntervention) => true,
                (IssueBeingFixed, IssueFixed) => true,
                (IssueBeingFixed, NeedsManualIntervention) => true,
                _ => false,
            }
        }

        pub async fn dismiss(&self, launch_id: &str) -> bool {
            let mut guard = self.states.write().await;
            guard.remove(launch_id).is_some()
        }

        pub async fn get_active(&self) -> Vec<LaunchStatusCard> {
            let guard = self.states.read().await;
            let mut cards: Vec<_> = guard.values().cloned().collect();
            cards.sort_by(|a, b| b.timestamp.cmp(&a.timestamp));
            cards
        }

        pub async fn prune(&self) {
            let now = Utc::now();
            let mut guard = self.states.write().await;
            // Drop stale (> STALE_AFTER_SECS old in terminal state)
            guard.retain(|_, card| {
                let age = (now - card.timestamp).num_seconds();
                age < STALE_AFTER_SECS
            });
            // Cap at MAX_ACTIVE_LAUNCHES
            if guard.len() > MAX_ACTIVE_LAUNCHES {
                let mut entries: Vec<_> = guard.iter().map(|(k, v)| (k.clone(), v.timestamp)).collect();
                entries.sort_by(|a, b| a.1.cmp(&b.1)); // oldest first
                let drop_count = guard.len() - MAX_ACTIVE_LAUNCHES;
                for (k, _) in entries.into_iter().take(drop_count) {
                    guard.remove(&k);
                }
            }
        }
    }

    impl Default for LaunchStateMachine {
        fn default() -> Self { Self::new() }
    ```
