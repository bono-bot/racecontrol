    **Step C — create `scripts/audit-handwritten-vs-generated.sh`:**

    Per D-12. This is the human-readable drift audit used in Plan 03 before flipping `src/index.ts` re-exports.

    ```bash
    #!/bin/bash
    # scripts/audit-handwritten-vs-generated.sh
    # D-12 drift audit: per hand-written .ts in src/, structural-diff against generated/ equivalent.
    set -euo pipefail

    GEN_DIR="packages/shared-types/generated"
    SRC_DIR="packages/shared-types/src"

    # Early exit if generated dir is empty (pre-Plan-02 state)
    if [ -z "$(ls -A $GEN_DIR 2&gt;/dev/null | grep -v '^\.')" ]; then
      echo "SKIPPED — no generated types yet, run Plan 02 first"
      exit 0
    fi

    # For each type-name in the WHITELIST, find both src/ and generated/ .ts files
    WHITELIST="$GEN_DIR/.whitelist.txt"
    echo "=== D-12 DRIFT AUDIT ==="
    while IFS= read -r TYPE_NAME; do
      [[ "$TYPE_NAME" =~ ^#.*$ ]] &amp;&amp; continue  # skip comments
      [[ -z "$TYPE_NAME" ]] &amp;&amp; continue
      # Find type in hand-written src/
      SRC_HIT=$(grep -l "^export type { .*\\b$TYPE_NAME\\b" "$SRC_DIR"/*.ts 2&gt;/dev/null || true)
      GEN_HIT="$GEN_DIR/$TYPE_NAME.ts"
      if [ -f "$GEN_HIT" ] &amp;&amp; [ -n "$SRC_HIT" ]; then
        echo "--- $TYPE_NAME ---"
        echo "  src: $SRC_HIT"
        echo "  gen: $GEN_HIT"
        # Normalize both: strip comments, blank lines, trailing whitespace
        SRC_NORM=$(grep -E "^\s*[a-z_]+.*;" "$SRC_HIT" | grep -v "^\s*//" | sort)
        GEN_NORM=$(grep -E "^\s*[a-z_]+.*;" "$GEN_HIT" | grep -v "^\s*//" | sort)
        if [ "$SRC_NORM" = "$GEN_NORM" ]; then
          echo "  STATUS: PARITY OK"
        else
          echo "  STATUS: DRIFT — fields differ; review before flipping re-export"
          diff &lt;(echo "$SRC_NORM") &lt;(echo "$GEN_NORM") || true
        fi
      fi
    done &lt; "$WHITELIST"
    ```

    - Exit 0 on success, exit 1 on any failure, never swallow errors with `|| true` except where noted.
