```rust
/// GLD-C-03: Receive CSV telemetry fallback from rc-agent at session end (D-07/D-09).
/// Staff-authenticated via sentry_service_key (NOT public). Max body 50MB.
async fn telemetry_fallback_handler(
    State(state): State<Arc<AppState>>,
    Path(session_id): Path<String>,
    headers: axum::http::HeaderMap,
    mut multipart: axum::extract::Multipart,
) -> axum::response::Response {
    // Service key gate (inline, per existing pattern)
    let expected = state.config.pods.sentry_service_key.as_deref().unwrap_or("");
    let provided = headers.get("X-Service-Key")
        .and_then(|v| v.to_str().ok())
        .unwrap_or("");
    if expected.is_empty() || provided.is_empty() || provided != expected {
        return (StatusCode::UNAUTHORIZED, "Invalid service key").into_response();
    }

    // Sanitize session_id (reject path traversal)
    if session_id.contains("..") || session_id.contains('/') || session_id.contains('\\') {
        return (StatusCode::BAD_REQUEST, "invalid session id").into_response();
    }

    // Extract the CSV part from multipart body
    let mut csv_bytes: Option<bytes::Bytes> = None;
    while let Ok(Some(field)) = multipart.next_field().await {
        if field.name() == Some("csv") {
            match field.bytes().await {
                Ok(b) => { csv_bytes = Some(b); break; }
                Err(e) => return (StatusCode::BAD_REQUEST, format!("read error: {e}")).into_response(),
            }
        }
    }
    let Some(bytes) = csv_bytes else {
        return (StatusCode::BAD_REQUEST, "missing 'csv' field").into_response();
    };

    // Write to C:\RacingPoint\telemetry-fallback\{session_id}.csv
    let dir = std::path::Path::new("C:\\RacingPoint\\telemetry-fallback");
    if let Err(e) = tokio::fs::create_dir_all(dir).await {
        tracing::error!(error = %e, "failed to create telemetry-fallback dir");
        return (StatusCode::INTERNAL_SERVER_ERROR, "mkdir failed").into_response();
    }
    let path = dir.join(format!("{session_id}.csv"));
    if let Err(e) = tokio::fs::write(&path, &bytes).await {
        tracing::error!(error = %e, path = %path.display(), "failed to write fallback csv");
        return (StatusCode::INTERNAL_SERVER_ERROR, "write failed").into_response();
    }

    // Update billing_sessions.csv_fallback_received_at
    let now_utc = chrono::Utc::now().to_rfc3339();
    let update_res = sqlx::query(
        "UPDATE billing_sessions SET csv_fallback_received_at = ? WHERE id = ?"
    )
    .bind(&now_utc)
    .bind(&session_id)
    .execute(&state.db)
    .await;

    match update_res {
        Ok(r) if r.rows_affected() > 0 => {
            tracing::info!(session_id = %session_id, bytes = bytes.len(), "csv fallback received");
            (StatusCode::OK, "received").into_response()
        }
        Ok(_) => (StatusCode::NOT_FOUND, "session not found").into_response(),
        Err(e) => {
            tracing::error!(error = %e, "csv fallback DB update failed");
            (StatusCode::INTERNAL_SERVER_ERROR, "db update failed").into_response()
        }
    }
}
```

```rust
use std::path::Path;
use std::time::Duration;

const CSV_PATH: &str = r"C:\RacingPoint\laps-offline.csv";
const MIN_CONTENT_BYTES: u64 = 10; // anything smaller is header-only or empty

/// GLD-C-03: POST the local CSV fallback file to the server after session end.
/// Read-then-clear ordering: buffer in memory → POST → clear ONLY on confirmed 200.
/// Retries with exponential backoff up to ~4 minutes total envelope.
pub async fn push_csv_fallback(
    server_url: String,
    service_key: String,
    session_id: String,
) -> anyhow::Result<()> {
    let path = Path::new(CSV_PATH);
    // Skip if file missing or too small
    let meta = match tokio::fs::metadata(path).await {
        Ok(m) => m,
        Err(_) => {
            tracing::debug!(path = CSV_PATH, "csv fallback absent — skipping push");
            return Ok(());
        }
    };
    if meta.len() < MIN_CONTENT_BYTES {
        tracing::debug!(bytes = meta.len(), "csv fallback empty — skipping push");
        return Ok(());
    }

    // Read INTO MEMORY before any network activity (pitfall 5: never clear before confirmed 200)
    let body_bytes = tokio::fs::read(path).await?;

    let url = format!("{}/api/v1/sessions/{}/telemetry-fallback",
        server_url.trim_end_matches('/'), session_id);
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(60))
        .build()?;

    // Exponential backoff: 2, 4, 8, 16, 32, 64, 128 = 254s total retry envelope
    let backoffs = [2u64, 4, 8, 16, 32, 64, 128];
    let mut last_err: Option<String> = None;

    for (attempt, delay) in std::iter::once(0u64).chain(backoffs.iter().copied()).enumerate() {
        if delay > 0 {
            tokio::time::sleep(Duration::from_secs(delay)).await;
        }
        let form = reqwest::multipart::Form::new()
            .part("csv", reqwest::multipart::Part::bytes(body_bytes.clone())
                .file_name(format!("{session_id}.csv"))
                .mime_str("text/csv")?);
        let res = client.post(&url)
            .header("X-Service-Key", &service_key)
            .multipart(form)
            .send()
            .await;
        match res {
            Ok(r) if r.status().is_success() => {
                tracing::info!(session_id = %session_id, bytes = body_bytes.len(),
                    "csv fallback pushed successfully");
                // ONLY NOW clear the local file
                if let Err(e) = clear_csv_laps().await {
                    tracing::warn!(error = %e, "clear_csv_laps failed after successful push");
                }
                return Ok(());
            }
            Ok(r) => {
                last_err = Some(format!("attempt {}: HTTP {}", attempt + 1, r.status()));
            }
            Err(e) => {
                last_err = Some(format!("attempt {}: {}", attempt + 1, e));
            }
        }
    }
    Err(anyhow::anyhow!("csv fallback push failed after retries: {}",
        last_err.unwrap_or_else(|| "unknown".into())))
}
```
