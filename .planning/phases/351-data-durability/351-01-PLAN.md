**Mod 1 — Add admin.db VACUUM INTO block in backup_tick() (OPS-08, per D-01/D-02):**

After the telemetry.db backup block (after line ~197), add an admin.db block. The admin.db is a SEPARATE SQLite file (not the same pool as state.db), so use a direct `sqlite3` CLI call via `tokio::process::Command` since we don't have a SqlxPool for admin.db:

```rust
// Backup admin.db if configured (OPS-08, Phase 351)
let admin_db_path = state.config.backup.admin_db_path.clone();
if !admin_db_path.is_empty() {
    let start = std::time::Instant::now();
    let admin_prefix = "admin";
    let daily_name = format!("{}-{}.db", admin_prefix, timestamp);
    let daily_path = format!("{}/{}", backup_dir, daily_name);
    // Use forward slashes for the destination path (SQLite on Windows)
    let sql_path = daily_path.replace('\\', "/");

    // admin.db is not in our SqlxPool — use sqlite3 CLI VACUUM INTO
    // This is WAL-safe and non-blocking, same as VACUUM INTO via sqlx.
    let vacuum_cmd = format!("VACUUM INTO '{}'", sql_path);
    let admin_src = admin_db_path.replace('\\', "/");
    let result = tokio::process::Command::new("sqlite3")
        .arg(&admin_src)
        .arg(&vacuum_cmd)
        .output()
        .await;

    match result {
        Ok(output) if output.status.success() => {
            let elapsed = start.elapsed().as_secs();
            tracing::info!(target: LOG_TARGET, "admin.db backup created: {} ({}s)", daily_name, elapsed);
            // Record size and update BackupStatus
            if let Ok(meta) = std::fs::metadata(&daily_path) {
                let size = meta.len();
                let ist_str = crate::whatsapp_alerter::ist_now_string();
                let mut status = state.backup_status.write().await;
                status.last_admin_backup_at = Some(ist_str);
                status.last_admin_backup_size = Some(size);
                // Zero-byte check (OPS-14, per D-06)
                if size == 0 {
                    drop(status); // release lock before .await
                    let msg = format!(
                        "[BACKUP ALERT] admin.db backup is 0 bytes — {} | {}",
                        daily_name,
                        crate::whatsapp_alerter::ist_now_string()
                    );
                    tracing::error!(target: LOG_TARGET, "{}", msg);
                    crate::whatsapp_alerter::send_whatsapp(&state.config, &msg).await;
                }
            }

            // Weekly snapshot on Sunday
            if is_weekly {
                let weekly_name = format!("{}-weekly-{}-W{:02}.db", admin_prefix, year, week_num);
                let weekly_path = format!("{}/{}", backup_dir, weekly_name);
                if let Err(e) = std::fs::copy(&daily_path, &weekly_path) {
                    tracing::warn!(target: LOG_TARGET, "Weekly copy for admin failed: {}", e);
                }
            }
        }
        Ok(output) => {
            let stderr = String::from_utf8_lossy(&output.stderr);
            tracing::error!(target: LOG_TARGET, "VACUUM INTO admin.db failed: {}", stderr);
        }
        Err(e) => {
            tracing::error!(target: LOG_TARGET, "sqlite3 command for admin.db failed: {}", e);
        }
    }

    rotate_backups(&backup_dir, admin_prefix, daily_retain, weekly_retain, monthly_retain)?;
    total_count += count_backup_files(&backup_dir, admin_prefix);
}
```

```rust
let monthly_pattern = format!("{}-monthly-", prefix);
```

```rust
// Step C: Transfer via rsync (preferred) or SCP (fallback), with 120s timeout.
let use_rsync = state.config.backup.use_rsync;
let transfer_result = if use_rsync {
    // rsync via Git Bash rsync on Windows
    let rsync_dest = format!("{}:{}/", remote_host, remote_path);
    tokio::time::timeout(
        Duration::from_secs(120),
        tokio::process::Command::new("rsync")
            .arg("-az")
            .arg("--checksum")
            .arg("--no-perms")
            .arg("-e")
            .arg("ssh -o StrictHostKeyChecking=no -o BatchMode=yes")
            .arg(backup_path)
            .arg(&rsync_dest)
            .output(),
    )
    .await
} else {
    // SCP fallback (existing behavior)
    let remote_dest = format!("{}:{}/{}", remote_host, remote_path, filename);
    tokio::time::timeout(
        Duration::from_secs(120),
        tokio::process::Command::new("scp")
            .arg("-o").arg("StrictHostKeyChecking=no")
            .arg("-o").arg("BatchMode=yes")
            .arg("-o").arg("ConnectTimeout=10")
            .arg(backup_path)
            .arg(&remote_dest)
            .output(),
    )
    .await
};

let transfer_output = match transfer_result {
    Err(_timeout) => {
        let tool = if use_rsync { "rsync" } else { "SCP" };
        let msg = format!("{} transfer timed out after 120s for {}", tool, filename);
        tracing::error!(target: LOG_TARGET, "{}", msg);
        let mut status = state.backup_status.write().await;
        status.remote_reachable = false;
        return Err(anyhow::anyhow!(msg));
    }
    Ok(Err(e)) => {
        let msg = format!("Transfer spawn error: {}", e);
        tracing::error!(target: LOG_TARGET, "{}", msg);
        let mut status = state.backup_status.write().await;
        status.remote_reachable = false;
        return Err(anyhow::anyhow!(msg));
    }
    Ok(Ok(output)) => output,
};

if !transfer_output.status.success() {
    let stderr = String::from_utf8_lossy(&transfer_output.stderr);
    let tool = if use_rsync { "rsync" } else { "SCP" };
    let msg = format!("{} transfer failed for {}: {}", tool, filename, stderr);
    tracing::error!(target: LOG_TARGET, "{}", msg);
    let mut status = state.backup_status.write().await;
    status.remote_reachable = false;
    return Err(anyhow::anyhow!(msg));
```
