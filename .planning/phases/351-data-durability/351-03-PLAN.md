**Step 1 — Create scripts/backup-cloud.sh:**

```bash
#!/usr/bin/env bash
# Phase 351 — Cloud backup script for Bono VPS
# Backs up cloud racecontrol.db + cloud admin.db using sqlite3 .backup
# Runs daily at 03:00 IST (21:30 UTC) via cron
# Stores backups in /root/backups/cloud/ with 30-day rotation + first-of-month 12-month retention

set -euo pipefail

BACKUP_ROOT="/root/backups/cloud"
TIMESTAMP=$(date -u -d '+5 hours 30 minutes' '+%Y-%m-%d_%H%M')
BACKUP_DIR="$BACKUP_ROOT/$TIMESTAMP"
FAILURES=0
MAX_DAILY=30
MAX_MONTHLY_MONTHS=12

# Database paths on Bono VPS
RC_DB="/root/racingpoint/racecontrol/data/racecontrol.db"
ADMIN_DB="/root/racingpoint/racingpoint-admin/data/admin.db"

mkdir -p "$BACKUP_DIR"

echo "=== Cloud Backup — $TIMESTAMP ==="

# Backup racecontrol.db (OPS-09)
if [ -f "$RC_DB" ]; then
    sqlite3 "$RC_DB" ".backup '$BACKUP_DIR/racecontrol.db'" 2>/dev/null && \
        echo "  OK: racecontrol.db ($(wc -c < "$BACKUP_DIR/racecontrol.db") bytes)" || \
        { echo "  FAIL: racecontrol.db backup command failed" >&2; FAILURES=$((FAILURES + 1)); }
else
    echo "  FAIL: racecontrol.db not found at $RC_DB" >&2
    FAILURES=$((FAILURES + 1))
fi

# Backup admin.db (OPS-08)
if [ -f "$ADMIN_DB" ]; then
    sqlite3 "$ADMIN_DB" ".backup '$BACKUP_DIR/admin.db'" 2>/dev/null && \
        echo "  OK: admin.db ($(wc -c < "$BACKUP_DIR/admin.db") bytes)" || \
        { echo "  FAIL: admin.db backup command failed" >&2; FAILURES=$((FAILURES + 1)); }
else
    echo "  WARN: admin.db not found at $ADMIN_DB — skipping"
fi

# Zero-byte validation (OPS-14)
for db in racecontrol.db admin.db; do
    DB_PATH="$BACKUP_DIR/$db"
    if [ -f "$DB_PATH" ] && [ ! -s "$DB_PATH" ]; then
        echo "  FAIL: $db is 0 bytes!" >&2
        FAILURES=$((FAILURES + 1))
    fi
done

# Prune old daily backups beyond MAX_DAILY, keeping first-of-month for 12 months
DIRS=$(ls -d "$BACKUP_ROOT"/20* 2>/dev/null | sort)
DIR_COUNT=$(echo "$DIRS" | grep -c '.' 2>/dev/null || echo 0)

if [ "$DIR_COUNT" -gt "$MAX_DAILY" ]; then
    echo ""
    echo "Pruning old backups (keeping last $MAX_DAILY + first-of-month)..."
    CUTOFF_DATE=$(date -u -d "-$MAX_MONTHLY_MONTHS months" '+%Y-%m-%d' 2>/dev/null || echo "")
    echo "$DIRS" | head -$((DIR_COUNT - MAX_DAILY)) | while read dir; do
        DIRNAME=$(basename "$dir")
        # Keep first-of-month snapshots (YYYY-MM-01_* pattern) for 12 months
        if [[ "$DIRNAME" == *"-01_"* ]]; then
            DIR_DATE="${DIRNAME%_*}"
            if [ -n "$CUTOFF_DATE" ] && [[ "$DIR_DATE" < "$CUTOFF_DATE" ]]; then
                rm -rf "$dir"
                echo "  Removed (>12 months): $DIRNAME"
            else
                echo "  Kept (first-of-month): $DIRNAME"
            fi
        else
            rm -rf "$dir"
            echo "  Removed: $DIRNAME"
        fi
    done
fi

echo ""
echo "=== Cloud Backup Complete — Failures: $FAILURES ==="
exit $FAILURES
```
