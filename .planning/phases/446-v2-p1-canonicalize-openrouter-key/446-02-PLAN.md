```js
// Phase 446 canonical-first dual-read. Canonical env name is OPENROUTER_KEY.
// OPENROUTER_API_KEY is a deprecated fallback that fires a one-shot console.warn.
// Local const name is kept as OPENROUTER_API_KEY because downstream usage
// (sanitizeOutput regex ~line 35, Authorization header ~line 69) references

const OPENROUTER_API_KEY = (() => {
  if (process.env.OPENROUTER_KEY && process.env.OPENROUTER_KEY.length > 0) {
    return process.env.OPENROUTER_KEY;
  }
  if (process.env.OPENROUTER_API_KEY && process.env.OPENROUTER_API_KEY.length > 0) {
    console.warn('[whatsapp-bot] OPENROUTER_API_KEY is deprecated — rename to OPENROUTER_KEY in pm2 env (read once, will not repeat)');
    return process.env.OPENROUTER_API_KEY;
  }
  return undefined;
})();
```
